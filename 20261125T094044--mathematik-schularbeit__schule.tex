\question[4] \textbf{Grundrechenarten:} Berechne das Ergebnis von $12 \cdot (4 + 3) - 18$.
\begin{solutionorgrid}[3cm]
$12 \cdot 7 - 18 = 84 - 18 = \mathbf{66}$
\end{solutionorgrid}

\question \textbf{Textaufgabe: Klassenfahrt} \\
Für eine Klassenfahrt fallen Gesamtkosten von 480 € an.
\begin{parts}
  \part[2] Berechne die Kosten pro Person, wenn 24 Schülerinnen und Schüler teilnehmen.
  \begin{solutionorlines}[2cm]
  $480\,\text{€} : 24 = \mathbf{20\,\text{€}}$ pro Person.
  \end{solutionorlines}

  \part[3] Wie viel müsste jede Person zahlen, wenn 4 Kinder krankheitsbedingt ausfallen?
  \begin{solutionorlines}[2.5cm]
  Bei $24 - 4 = 20$ Teilnehmenden: $480\,\text{€} : 20 = \mathbf{24\,\text{€}}$ pro Person.
  \end{solutionorlines}
\end{parts}
